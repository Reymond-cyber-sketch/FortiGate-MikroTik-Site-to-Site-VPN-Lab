COMANDOS PRINCIPALES UTILIZADOS

/ip address add address=203.0.113.98/30 interface=ether1 comment="WAN hacia ISP"

/ip address add address=10.24.96.129/28 interface=ether2 comment="LAN WEB"

/ip route add dst-address=0.0.0.0/0 gateway=203.0.113.97

/ip dns set servers=8.8.8.8,1.1.1.1 allow-remote-requests=yes

/ip firewall nat add chain=srcnat src-address=10.24.96.128/28 dst-address=10.24.96.0/25 action=accept comment="NO NAT hacia VPN"

/ip firewall nat add chain=srcnat src-address=10.24.96.128/28 out-interface=ether1 action=masquerade comment="NAT WEB hacia Internet"

/ip ipsec profile add name=FGT-IKE2 hash-algorithm=sha256 enc-algorithm=des dh-group=modp2048

/ip ipsec proposal add name=FGT-P2 auth-algorithms=sha256 enc-algorithms=des pfs-group=none

/ip ipsec peer add name=FGT-USER address=198.51.100.98/32 exchange-mode=ike2 profile=FGT-IKE2

/ip ipsec policy add src-address=10.24.96.128/28 dst-address=10.24.96.0/25 tunnel=yes action=encrypt peer=FGT-USER proposal=FGT-P2
